/* Standalone export: rewrite JS-referenced asset paths to the bundled blob URLs.
   Image paths inside this design are built as strings in component logic
   (background-image url(...), <img src>), so the bundler cannot see them. */
(function () {
  function map() { return window.__resources || null; }
  function fix(el) {
    var r = map(); if (!r) return;
    if (el.tagName === 'IMG') {
      var s = el.getAttribute('src');
      if (s && r[s]) el.src = r[s];
    }
    var bg = el.style && el.style.backgroundImage;
    if (bg && bg.indexOf('url(') > -1) {
      var out = bg.replace(/url\((['"]?)([^'")]+)\1\)/g, function (m, q, p) {
        return r[p] ? 'url("' + r[p] + '")' : m;
      });
      if (out !== bg) el.style.backgroundImage = out;
    }
  }
  function sweep(root) {
    if (!map()) return;
    fix(root);
    var all = root.querySelectorAll ? root.querySelectorAll('img,[style*="url("]') : [];
    for (var i = 0; i < all.length; i++) fix(all[i]);
  }
  function start() {
    sweep(document.documentElement);
    new MutationObserver(function (recs) {
      for (var i = 0; i < recs.length; i++) {
        var rec = recs[i];
        if (rec.type === 'attributes') fix(rec.target);
        for (var j = 0; j < rec.addedNodes.length; j++) {
          var n = rec.addedNodes[j];
          if (n.nodeType === 1) sweep(n);
        }
      }
    }).observe(document.documentElement, { subtree: true, childList: true, attributes: true, attributeFilter: ['src', 'style'] });
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', start);
  else start();
})();
</script>
